cmake_minimum_required(VERSION 3.27)
project(torch LANGUAGES CXX)
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SPUTNIKTORCH_ERROR_NAMESPACE "Torch.Errors" CACHE STRING "Sputnik native error namespace")
add_executable(torch-run tools/launcher.cpp)
# The adapter calls prebuilt LibTorch kernels. An explicit C++ provider avoids
# making nvcc, hipcc or language bindings consumer build dependencies.
set(SPUTNIKTORCH_PREBUILT_TORCH_ROOT "" CACHE PATH "Explicit Linux LibTorch C++ provider")
if(SPUTNIKTORCH_PREBUILT_TORCH_ROOT)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Explicit prebuilt Torch provider is supported on Linux")
  endif()
  include(cmake/TorchProvider.cmake)
  torch_provider("${SPUTNIKTORCH_PREBUILT_TORCH_ROOT}")
  set(Torch_VERSION "${TORCH_PROVIDER_VERSION}")
  set(TORCH_INCLUDE_DIRS "${SPUTNIKTORCH_PREBUILT_TORCH_ROOT}/include" "${SPUTNIKTORCH_PREBUILT_TORCH_ROOT}/include/torch/csrc/api/include")
  include_directories(SYSTEM ${TORCH_INCLUDE_DIRS})
  set(TORCH_LIBRARIES ${TORCH_PROVIDER_LIBRARIES})
  add_compile_definitions(_GLIBCXX_USE_CXX11_ABI=${TORCH_PROVIDER_ABI})
else()
  find_package(Torch 2.14.0 CONFIG REQUIRED)
endif()
if(NOT Torch_VERSION VERSION_EQUAL "2.14.0" AND NOT Torch_VERSION VERSION_EQUAL "2.14.1")
  message(FATAL_ERROR "This torch adapter requires LibTorch 2.14.0 or 2.14.1; found '${Torch_VERSION}'")
endif()
list(APPEND TORCH_LIBRARIES Threads::Threads)
set(SPUTNIK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../sputnik-lang" CACHE PATH "Sputnik source root")
set(GLOO_INCLUDE_DIR "" CACHE PATH "Gloo source root matching the LibTorch build")
set(ONNXRUNTIME_ROOT "" CACHE PATH "Optional C++ ONNX Runtime for export verification")
file(GLOB_RECURSE ADAPTER_SOURCES CONFIGURE_DEPENDS native/*.cpp)
add_library(torch_adapter OBJECT ${ADAPTER_SOURCES})
set_target_properties(torch_adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(torch_adapter PRIVATE "${SPUTNIK_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/native" ${TORCH_INCLUDE_DIRS})
target_link_libraries(torch_adapter PRIVATE ${TORCH_LIBRARIES})
target_compile_definitions(torch_adapter PRIVATE SPUTNIKTORCH_ERROR_NAMESPACE="${SPUTNIKTORCH_ERROR_NAMESPACE}")
if(SPUTNIKTORCH_PREBUILT_TORCH_ROOT)
  # The official Linux C++ distribution namespaces its bundled ONNX types.
  set_source_files_properties(native/onnx/trace.cpp PROPERTIES COMPILE_DEFINITIONS ONNX_NAMESPACE=onnx_torch)
endif()
# A prepared C-ABI artifact for isolated notebook workers. It contains no
# Sputnik algorithms; the host links the original package modules at Run time.
add_library(torch_notebook MODULE tools/notebook_abi.cpp $<TARGET_OBJECTS:torch_adapter>)
target_include_directories(torch_notebook PRIVATE "${SPUTNIK_ROOT}")
target_link_libraries(torch_notebook PRIVATE ${TORCH_LIBRARIES})
add_library(torch_package SHARED tools/notebook_abi.cpp $<TARGET_OBJECTS:torch_adapter>)
target_include_directories(torch_package PRIVATE "${SPUTNIK_ROOT}")
target_link_libraries(torch_package PRIVATE ${TORCH_LIBRARIES})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_target_properties(torch_package PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/vendor")
endif()
if(APPLE)
  target_link_options(torch_notebook PRIVATE "-undefined" "dynamic_lookup")
  target_link_options(torch_package PRIVATE "-undefined" "dynamic_lookup")
endif()
add_executable(torch_probe tests/native_probe.cpp)
target_include_directories(torch_probe PRIVATE "${SPUTNIK_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(torch_probe PRIVATE ${TORCH_LIBRARIES})
add_executable(torch_checkpoint_fixtures tests/checkpoint_fixtures.cpp)
target_link_libraries(torch_checkpoint_fixtures PRIVATE ${TORCH_LIBRARIES})
if(ONNXRUNTIME_ROOT)
  add_executable(torch_onnx_runtime tests/onnx_runtime.cpp)
  target_include_directories(torch_onnx_runtime PRIVATE "${ONNXRUNTIME_ROOT}/include")
  target_link_libraries(torch_onnx_runtime PRIVATE ${TORCH_LIBRARIES} "${ONNXRUNTIME_ROOT}/lib/libonnxruntime.dylib")
  set_target_properties(torch_onnx_runtime PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
endif()
if(GLOO_INCLUDE_DIR)
  target_include_directories(torch_adapter PRIVATE "${GLOO_INCLUDE_DIR}")
  target_compile_definitions(torch_adapter PRIVATE USE_C10D_GLOO=1)
  add_executable(torch_gloo_probe tests/gloo_probe.cpp)
  target_include_directories(torch_gloo_probe PRIVATE "${GLOO_INCLUDE_DIR}")
  target_compile_definitions(torch_gloo_probe PRIVATE USE_C10D_GLOO=1)
  target_link_libraries(torch_gloo_probe PRIVATE ${TORCH_LIBRARIES})
endif()
enable_testing()
add_test(NAME cpu COMMAND torch_probe cpu)
if(SPUTNIKTORCH_PREBUILT_TORCH_ROOT AND TORCH_PROVIDER_BACKEND STREQUAL "rocm")
  add_test(NAME rocm COMMAND torch_probe cuda:0)
endif()
add_executable(torch_execution_mode_test tests/execution_mode.cpp)
target_include_directories(torch_execution_mode_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(torch_execution_mode_test PRIVATE ${TORCH_LIBRARIES})
add_test(NAME execution_mode COMMAND torch_execution_mode_test)
