cmake_minimum_required(VERSION 3.27)
project(sputniktorch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(sputniktorch-run tools/launcher.cpp)
find_package(Torch CONFIG REQUIRED)
set(SPUTNIK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../sputnik-lang" CACHE PATH "Sputnik source root")
set(GLOO_INCLUDE_DIR "" CACHE PATH "Gloo source root matching the LibTorch build")
set(ONNXRUNTIME_ROOT "" CACHE PATH "Optional C++ ONNX Runtime for export verification")
file(GLOB_RECURSE ADAPTER_SOURCES CONFIGURE_DEPENDS native/*.cpp)
add_library(sputniktorch_adapter OBJECT ${ADAPTER_SOURCES})
set_target_properties(sputniktorch_adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sputniktorch_adapter PRIVATE "${SPUTNIK_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/native" ${TORCH_INCLUDE_DIRS})
target_link_libraries(sputniktorch_adapter PRIVATE ${TORCH_LIBRARIES})
# A prepared C-ABI artifact for isolated notebook workers. It contains no
# Sputnik algorithms; the host links the original package modules at Run time.
add_library(sputniktorch_notebook MODULE tools/notebook_abi.cpp $<TARGET_OBJECTS:sputniktorch_adapter>)
target_include_directories(sputniktorch_notebook PRIVATE "${SPUTNIK_ROOT}")
target_link_libraries(sputniktorch_notebook PRIVATE ${TORCH_LIBRARIES})
if(APPLE)
  target_link_options(sputniktorch_notebook PRIVATE "-undefined" "dynamic_lookup")
endif()
add_executable(sputniktorch_probe tests/native_probe.cpp)
target_link_libraries(sputniktorch_probe PRIVATE ${TORCH_LIBRARIES})
add_executable(sputniktorch_checkpoint_fixtures tests/checkpoint_fixtures.cpp)
target_link_libraries(sputniktorch_checkpoint_fixtures PRIVATE ${TORCH_LIBRARIES})
if(ONNXRUNTIME_ROOT)
  add_executable(sputniktorch_onnx_runtime tests/onnx_runtime.cpp)
  target_include_directories(sputniktorch_onnx_runtime PRIVATE "${ONNXRUNTIME_ROOT}/include")
  target_link_libraries(sputniktorch_onnx_runtime PRIVATE ${TORCH_LIBRARIES} "${ONNXRUNTIME_ROOT}/lib/libonnxruntime.dylib")
  set_target_properties(sputniktorch_onnx_runtime PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
endif()
if(GLOO_INCLUDE_DIR)
  target_include_directories(sputniktorch_adapter PRIVATE "${GLOO_INCLUDE_DIR}")
  target_compile_definitions(sputniktorch_adapter PRIVATE USE_C10D_GLOO=1)
  add_executable(sputniktorch_gloo_probe tests/gloo_probe.cpp)
  target_include_directories(sputniktorch_gloo_probe PRIVATE "${GLOO_INCLUDE_DIR}")
  target_compile_definitions(sputniktorch_gloo_probe PRIVATE USE_C10D_GLOO=1)
  target_link_libraries(sputniktorch_gloo_probe PRIVATE ${TORCH_LIBRARIES})
endif()
enable_testing()
add_test(NAME cpu COMMAND sputniktorch_probe cpu)
